cmake_minimum_required(VERSION 3.25)
project(black_host_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ZYDIS_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ZYDIS_BUILD_DOXYGEN OFF CACHE BOOL "" FORCE)
add_subdirectory("${ZYDIS_SOURCE_DIR}" zydis)
add_subdirectory("${TOMLPLUSPLUS_SOURCE_DIR}" tomlplusplus)
file(GLOB sources CONFIGURE_DEPENDS "${XBE_SDK_DIR}/tools/recomp/*.cpp")
add_executable(xbe_recomp ${sources})
target_include_directories(xbe_recomp PRIVATE "${XBE_SDK_DIR}/runtime" "${XBE_SDK_DIR}/tools/recomp")
target_link_libraries(xbe_recomp PRIVATE Zydis tomlplusplus::tomlplusplus)
